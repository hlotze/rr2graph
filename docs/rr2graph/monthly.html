rr2graph.monthly

Monthly visualization generation pipeline for rr2graph.

This module contains the high-level plotting workflow used to generate monthly RR visualization dashboards.

Responsibilities include: - monthly dataframe filtering - visualization dispatching - export path generation - plot row composition - multi-format figure export

The module acts as the bridge between the low-level plotting modules and the orchestration layer.

 54def _ensure_output_dirs(base_dir: str | Path) -> dict[str, Path]:
 55    """
 56    Ensure export subdirectories exist.
 57
 58    Creates normalized output directories for all supported export
 59    formats.
 60
 61    Args:
 62        base_dir:
 63            Base export directory.
 64
 65    Returns:
 66        dict[str, Path]:
 67            Mapping of export format names to output paths.
 68    """
 69    base = Path(base_dir)
 70    dirs = {
 71        "png": base / "png",
 72        "pdf": base / "pdf",
 73        "svg": base / "svg",
 74    }
 75    for d in dirs.values():
 76        d.mkdir(parents=True, exist_ok=True)
 77    return dirs

102def _filter_last_months(
103    df: pd.DataFrame, date_col: str, num_months: int
104) -> pd.DataFrame:
105    """
106    Filter a dataframe to the most recent months.
107
108    Args:
109        df:
110            Source dataframe.
111
112        date_col:
113            Datetime column name.
114
115        num_months:
116            Number of months to retain.
117
118    Returns:
119        pd.DataFrame:
120            Filtered dataframe.
121    """
122    end_date = df[date_col].max()
123    start_date = end_date - pd.DateOffset(months=num_months)
124    return df[(df[date_col] >= start_date) & (df[date_col] <= end_date)]

127def _extract_months(df_heart: pd.DataFrame, num_months: int) -> list[pd.Period]:
128    """
129    Extract normalized monthly periods from RR data.
130
131    Args:
132        df_heart:
133            RR measurement dataframe.
134
135        num_months:
136            Maximum number of months to return.
137
138    Returns:
139        list[pd.Period]:
140            Sorted list of monthly periods.
141    """
142    months = df_heart["date_time"].dt.to_period("M").sort_values().unique()
143    return list(months[-num_months:])

276def gen_req_plot_type(
277    plot_type: str,
278    num_of_months: int,
279    df_heart: pd.DataFrame,
280    df_weight: pd.DataFrame,
281    axs: Axes | Sequence[Axes],
282) -> str:
283    """
284    Generate the requested monthly visualization strategy.
285
286    The function dispatches the requested plot type to the appropriate
287    monthly row generator implementation.
288
289    Supported plot types:
290        - histogram
291        - violin
292        - box_swarm
293
294    Args:
295        plot_type:
296            Requested visualization type.
297
298        num_of_months:
299            Number of months to visualize.
300
301        df_heart:
302            RR measurement dataframe.
303
304        df_weight:
305            Weight measurement dataframe.
306
307        axs:
308            Matplotlib axis container.
309
310    Returns:
311        str:
312            Generated export filename.
313
314    Raises:
315        KeyError:
316            Raised when an unsupported plot type is requested.
317    """
318    # Map plot identifiers to monthly row generator implementations.
319    dispatch: dict[str, RowFunc] = {
320        "histogram": gen_row_of_3_grapics_for_one_month_histo,
321        "violin": gen_row_of_3_grapics_for_one_month_violin,
322        "box_swarm": gen_row_of_3_grapics_for_one_month_box_swarm,
323    }
324
325    if plot_type not in dispatch:
326        raise KeyError(f"Unknown plot_type: {plot_type}")
327
328    row_func = dispatch[plot_type]
329
330    # Restrict datasets to the configured monthly time window.
331    df_heart = _filter_last_months(df_heart, "date_time", num_of_months)
332    df_weight = _filter_last_months(df_weight, "date", num_of_months)[
333        ["date", "weight"]
334    ]
335
336    # Extract all available monthly periods.
337    months = _extract_months(df_heart, num_of_months)
338
339    if len(months) == 1:
340        month = months[0]
341        row_func(month, axs, df_heart, df_weight)
342        start_month = month.start_time
343        return f"({start_month.strftime('%Y-%m')}) per month data and {plot_type}"
344
345    # Generate one visualization row per month.
346    for ax, month in zip(axs, months):
347        row_func(month, ax, df_heart, df_weight)
348
349    start_month = months[0].start_time
350    end_month = months[-1].end_time
351
352    return (
353        f"({start_month.strftime('%Y-%m')}__"
354        f"{end_month.strftime('%Y-%m')} {len(months)} months) "
355        f"per month data and {plot_type}"
356    )

364def generate_monthly_plots(
365    plot_type: str,
366    num_of_months: int,
367    df_heart,
368    df_weight,
369    out_dir: str | Path,
370) -> list[str]:
371    """
372    Generate and export monthly RR visualizations.
373
374    This function represents the high-level monthly rendering pipeline
375    used by the rr2graph orchestration layer.
376
377    Workflow steps:
378        1. Allocate figure and axes
379        2. Generate requested visualization type
380        3. Create export directories
381        4. Export figures to all supported formats
382        5. Release matplotlib resources
383
384    Args:
385        plot_type:
386            Requested visualization type.
387
388        num_of_months:
389            Number of months to visualize.
390
391        df_heart:
392            RR measurement dataframe.
393
394        df_weight:
395            Weight measurement dataframe.
396
397        out_dir:
398            Base export directory.
399
400    Returns:
401        list[str]:
402            List of generated export file paths.
403
404    Raises:
405        RuntimeError:
406            Raised when matplotlib rendering fails.
407
408        OSError:
409            Raised when export operations fail.
410    """
411    # Allocate figure and axes layout.
412    fig, axs = get_needed_fig_and_axs_array(num_of_months)
413
414    # Generate the requested visualization strategy.
415    filename = gen_req_plot_type(
416        plot_type=plot_type,
417        num_of_months=num_of_months,
418        df_heart=df_heart,
419        df_weight=df_weight,
420        axs=axs,
421    )
422
423    # Prepare export directory structure and file paths.
424    subdirs = _ensure_output_dirs(out_dir)
425    paths = _build_output_paths(subdirs, filename)
426
427    # Export the rendered figure into all configured formats.
428    for fmt, path in paths.items():
429        fig.savefig(path, format=fmt)
430
431    # Release matplotlib resources.
432    plt.close(fig)
433
434    return [str(p) for p in paths.values()]
